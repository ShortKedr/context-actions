cmake_minimum_required(VERSION 3.25)
project(ContextActions VERSION 0.1.0 LANGUAGES CXX RC)
if(NOT WIN32 OR NOT MSVC)
  message(FATAL_ERROR "Context Actions requires MSVC and Windows SDK (x64).")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_compile_options(/W4 /utf-8 /permissive- /EHsc)
add_compile_definitions(UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
add_library(actions_core STATIC src/core.cpp src/pdf.cpp src/office.cpp src/locale.cpp)
target_include_directories(actions_core PUBLIC src)
target_link_libraries(actions_core PUBLIC ole32 oleaut32 shell32 shlwapi windowscodecs uuid)
add_executable(ContextActions WIN32 src/main.cpp src/ui.cpp resources/app.rc)
target_link_libraries(ContextActions PRIVATE actions_core comctl32 comdlg32 dwmapi uxtheme)
target_link_options(ContextActions PRIVATE "/MANIFESTINPUT:${CMAKE_CURRENT_SOURCE_DIR}/resources/app.manifest")
add_library(ContextActionsShell SHARED src/shell.cpp resources/shell.def resources/app.rc)
target_link_libraries(ContextActionsShell PRIVATE actions_core)
add_executable(actions_tests tests/core_tests.cpp src/ui.cpp)
target_link_libraries(actions_tests PRIVATE actions_core comctl32 comdlg32 dwmapi uxtheme)
enable_testing()
add_test(NAME core COMMAND actions_tests "${CMAKE_BINARY_DIR}/test-data")
install(TARGETS ContextActions ContextActionsShell RUNTIME DESTINATION . LIBRARY DESTINATION .)
install(FILES docs/support.md docs/third-party.md DESTINATION docs)
