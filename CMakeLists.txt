cmake_minimum_required(VERSION 3.25)
if(CMAKE_HOST_SYSTEM_NAME STREQUAL "Darwin")
  set(CMAKE_OSX_ARCHITECTURES "arm64;x86_64" CACHE STRING "macOS architectures")
  set(CMAKE_OSX_DEPLOYMENT_TARGET "11.0" CACHE STRING "Minimum macOS version")
endif()
project(ContextActions VERSION 0.1.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

add_library(actions_core STATIC src/core.cpp src/command_line.cpp src/pdf.cpp src/locale.cpp)
target_include_directories(actions_core PUBLIC src)

if(WIN32)
  if(NOT MSVC)
    message(FATAL_ERROR "The Windows build requires MSVC and the Windows SDK.")
  endif()
  enable_language(RC)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  set_property(TARGET actions_core PROPERTY MSVC_RUNTIME_LIBRARY "${CMAKE_MSVC_RUNTIME_LIBRARY}")
  target_compile_options(actions_core PUBLIC /W4 /utf-8 /permissive- /EHsc)
  target_compile_options(actions_core PRIVATE "/d1trimfile:${CMAKE_CURRENT_SOURCE_DIR}")
  target_compile_definitions(actions_core PUBLIC UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
  target_sources(actions_core PRIVATE src/windows/platform.cpp src/windows/office.cpp)
  target_link_libraries(actions_core PUBLIC ole32 oleaut32 shell32 shlwapi uuid)
  # scripts/build-ui.ps1 builds the WinUI host using the native XAML MSBuild targets.
  add_library(ContextActionsShell SHARED src/shell.cpp resources/shell.def resources/app.rc)
  target_link_libraries(ContextActionsShell PRIVATE actions_core)
  target_compile_options(ContextActionsShell PRIVATE "/d1trimfile:${CMAKE_CURRENT_SOURCE_DIR}")
  target_link_options(ContextActionsShell PRIVATE /PDBALTPATH:ContextActionsShell.pdb)
  install(TARGETS ContextActionsShell RUNTIME DESTINATION . LIBRARY DESTINATION .)
elseif(APPLE OR CMAKE_SYSTEM_NAME STREQUAL "Linux")
  find_package(Threads REQUIRED)
  target_sources(actions_core PRIVATE src/posix/platform.cpp src/posix/office.cpp)
  target_link_libraries(actions_core PUBLIC Threads::Threads)
  target_compile_options(actions_core PRIVATE -Wall -Wextra -Wpedantic)
  if(APPLE)
    enable_language(OBJCXX)
    find_library(CORE_FOUNDATION CoreFoundation REQUIRED)
    target_link_libraries(actions_core PRIVATE "${CORE_FOUNDATION}")
    add_subdirectory(src/macos)
  else()
    add_subdirectory(src/linux)
  endif()
else()
  message(FATAL_ERROR "Context Actions supports Windows, macOS and Linux.")
endif()

enable_testing()
add_executable(actions_tests tests/core_tests.cpp)
target_link_libraries(actions_tests PRIVATE actions_core)
add_test(NAME core COMMAND actions_tests "${CMAKE_BINARY_DIR}/test-data")
if(WIN32)
  add_executable(shell_tests tests/shell_tests.cpp)
  target_link_libraries(shell_tests PRIVATE actions_core)
  add_dependencies(shell_tests ContextActionsShell)
  add_test(NAME shell_contract COMMAND shell_tests "$<TARGET_FILE:ContextActionsShell>" "${CMAKE_BINARY_DIR}/shell-test-data")
  install(FILES LICENSE docs/support.md docs/third-party.md DESTINATION docs)
elseif(CMAKE_SYSTEM_NAME STREQUAL "Linux")
  include(GNUInstallDirs)
  install(FILES LICENSE docs/support.md docs/third-party.md
    DESTINATION "${CMAKE_INSTALL_DATADIR}/doc/context-actions")
  include(packaging/linux/CPack.cmake)
endif()
